# Simulation method:

#        With the USER option the user can set the relation between flux and size of the input

SIM_SRCS_SIGMA_MIN 3.0

# Maximum flux of the simulated sources [in units of underlying noise sigma]
SIM_SRCS_SIGMA_MAX 100000.0

# Simulated number counts slope, considering dN/dS ~ S^-alpha
# Larger values of SIM_POWER_LAW_IDX implies a larger amount
# of faint sources
# - Euclidean non-evolving Universe (field galaxies)--> 2.5
# - cluster field --> ~ 1.5 - 2.0
SIM_POWER_LAW_IDX 1.7 # 2.5

# Minimum size of the injected sources in units of the PSF (1.0 = input PSF)
SIM_SRCS_SIZE_MIN 1.0

# Maximum size factor (e.g. 3x broader than input PSF)
SIM_SRCS_SIZE_MAX 20

#----------------------------------------------------------------------
# FLux - size relation
# S = S_min + (S_max-S_min) x {[(F-F_min)/(F_max-F_min)]^beta} x scatter
# with scatter=exp(0 ± sigma)
#----------------------------------------------------------------------

# Correlation strength beta (higher --> stronger flux-size link). It controls the
#  systematic trend: how strongly size grows with flux on average.
#  0 --> no trend, all sources have nearly the same size.
#  1 --> sizes scale steeply with flux.
SIM_SRCS_BETA 0.5

# Sigma in the lognormal scatter. Controls the spread around that trend at fixed flux.
#  Small sigma (~0.1) --> tight relation (all bright sources have nearly the same size).
#  Large sigma (~2.0) --> flux-size relation almost meaningles
#  In log10 terms: 0.3/ln⁡(10)≈0.130.3/ln(10)≈0.13 dex scatter.
#  In log10 terms: 1.0/ln⁡(10)≈0.431.0/ln(10)≈0.43 dex scatter.
SIM_SRCS_SIGMA 0.6
